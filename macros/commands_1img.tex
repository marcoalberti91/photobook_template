% Command for 1 landscape image with normal margins
\newcommand{\landscapeLargeMargin}[1]{%
    \begin{figure}[h]
        \centering
        \includegraphics[width=\textwidth]{#1}
    \end{figure}
}

% Command for 1 landscape image with normal margins and centered
\newcommand{\landscapeLargeMarginCentered}[1]{%
    \clearpage
    \thispagestyle{empty}
    \vspace*{\fill}
    \begin{center}
        \includegraphics[width=\textwidth,keepaspectratio]{#1}
    \end{center}
    \vspace*{\fill}
    \clearpage
}
\newcommand{\landscapeThinMargin}[1]{%
    \newgeometry{margin=1.1cm}
    \begin{figure}[h]
        \centering
        \includegraphics[width=\textwidth]{#1}
    \end{figure}
    \restoregeometry
}

% Command for 1 landscape image with reduced margins and centered
\newcommand{\landscapeThinMarginCentered}[1]{%
    \clearpage
    \newgeometry{margin=1.1cm}
    \thispagestyle{empty}
    \vspace*{\fill}
    \begin{center}
        \includegraphics[width=\textwidth,keepaspectratio]{#1}
    \end{center}
    \vspace*{\fill}
    \restoregeometry
    \clearpage
}

% Command for 1 fullscreen portrait image
\newcommand{\portraitFullscreen}[1]{%
    \newgeometry{margin=0cm}
    \begin{figure}
        \centering
        \includegraphics[width=\paperwidth,height=\paperheight]{#1}
    \end{figure}
    \restoregeometry
}

% Left half: cropped, shifted 6mm left
\newcommand{\splitImageLeft}[2][0pt]{%
  \clearpage
  \newgeometry{margin=#1}%
  \thispagestyle{empty}%
  \noindent\hspace*{\fill}%
  \adjincludegraphics[
    width=\paperwidth,
    height=\paperheight,
    keepaspectratio,
    trim={12mm 0 {.5\width-12mm} 0}, % Shift crop window 6mm left
    clip
  ]{#2}%
  \par
  \restoregeometry
}

% Right half: cropped, shifted 6mm right
\newcommand{\splitImageRight}[2][0pt]{%
  \clearpage
  \newgeometry{margin=#1}%
  \thispagestyle{empty}%
  \noindent
  \adjincludegraphics[
    width=\paperwidth,
    height=\paperheight,
    keepaspectratio,
    trim={{.5\width-12mm} 0 12mm 0}, % Shift crop window 6mm right
    clip
  ]{#2}%
  \hspace*{\fill}%
  \par
  \restoregeometry
}

% Command for 1 portrait image occupying half page
\newcommand{\portraitHalfPage}[1]{%
    \begin{figure}[h]
        \centering
        \includegraphics[width=0.55\textwidth,keepaspectratio]{#1}
    \end{figure}
}
